simulator.py (pure logic, testable)
import random
from datetime import datetime

def generate_raw():
    return {
        "timestamp": datetime.utcnow().isoformat(),
        "temperature": round(random.normalvariate(25, 2), 2),
        "humidity": round(random.normalvariate(55, 5), 2),
        "motion": random.random() < 0.1,
        "light": round(random.uniform(50, 900), 2)
    }

def add_features(data):
    # Example derived features
    data["heat_index"] = round(
        data["temperature"] + 0.1 * data["humidity"], 2
    )

    data["is_dark"] = data["light"] < 200

    return data

mqtt_client.py
import json
import paho.mqtt.client as mqtt

class MQTTClient:
    def __init__(self, host, port, topic):
        self.client = mqtt.Client()
        self.host = host
        self.port = port
        self.topic = topic

    def connect(self):
        self.client.connect(self.host, self.port, 60)
        self.client.loop_start()

    def publish(self, payload):
        self.client.publish(self.topic, json.dumps(payload))

    def close(self):
        self.client.loop_stop()
        self.client.disconnect()
app.py
import time
import uuid

from simulator import generate_raw
from feature_engine import add_features
from mqtt_client import MQTTClient

DEVICE_ID = str(uuid.uuid4())[:8]

mqtt_client = MQTTClient("mqtt-broker", 1883, "aiot/sensors/telemetry")
mqtt_client.connect()

while True:
    raw = generate_raw()
    enriched = add_features(raw)
    enriched["device_id"] = DEVICE_ID

    mqtt_client.publish(enriched)
    print(enriched)

    time.sleep(1)
